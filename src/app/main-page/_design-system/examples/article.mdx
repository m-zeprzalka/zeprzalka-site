```tsx
"use client"

import { useState } from "react"
import { Button } from "@/app/main-page/_components/ui/button"

/** Licznik — stan lokalny i obsługa kliknięcia. */
export function Counter({ start = 0 }: { start?: number }) {
  const [count, setCount] = useState(start)
  return (
    <Button variant="outline" onClick={() => setCount((n) => n + 1)}>
      Kliknięto {count} razy
    </Button>
  )
}
```

```ts
type Locale = "pl" | "en"

// Strefa UTC: data wpisu nie przeskakuje o dzień, niezależnie od strefy czasowej czytelnika ani serwera.
export function formatDate(iso: string, locale: Locale = "pl"): string {
  return new Intl.DateTimeFormat(locale, { dateStyle: "long", timeZone: "UTC" }).format(new Date(iso))
}
```
